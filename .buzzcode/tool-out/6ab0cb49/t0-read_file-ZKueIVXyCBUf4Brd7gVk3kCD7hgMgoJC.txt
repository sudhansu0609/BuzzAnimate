  1	//! The tool set, its shortcuts, and how the toolbar is grouped.
  2	//!
  3	//! This module is the *catalogue*: what tools exist, what they are called, what
  4	//! key selects them, and how they group in the toolbar. The behaviour lives in
  5	//! `buzz-app`, so the shortcut map can be tested without a window.
  6	//!
  7	//! Letters follow Animate exactly. Muscle memory is the whole point — a user
  8	//! who presses `V` expects the Selection tool, and getting that wrong is worse
  9	//! than having fewer tools.
 10

 46	/// Whether a tool is usable yet.
 47	///
 48	/// Shown honestly in the toolbar. A tool that looks available but silently does
 49	/// nothing is worse than one that says it is not ready.
 50	#[derive(Debug, Clone, Copy, PartialEq, Eq)]
 51	pub enum ToolStatus {
 52	    Ready,
 53	    /// Arrives in a later phase; the tooltip says which.
 54	    Planned(&'static str),
 55	}

 90	    pub fn shortcut(self) -> Option<Key> {
 91	        use ToolId::*;
 92	        match self {
 93	            Selection => Some(Key::V),
 94	            Subselection => Some(Key::A),
 95	            Lasso => Some(Key::L),
 96	            // Animate has no letter for the Magic Wand — it is a mode of the

 99	            MagicWand => Some(Key::G),
100	            FreeTransform => Some(Key::Q),
101	            Pen => Some(Key::P),
102	            Text => Some(Key::T),
103	            Line => Some(Key::N),
104	            Rectangle => Some(Key::R),
105	            Oval => Some(Key::O),
106	            Pencil => Some(Key::Y),
107	            Brush => Some(Key::B),
108	            Bone => Some(Key::M),
109	            // Animate's own letter for Asset Warp.
110	            AssetWarp => Some(Key::W),
111	            // Animate has no motion-path tool; J is free and next to nothing
112	            // that would be confused with it.
113	            MotionPath => Some(Key::J),

115	            InkBottle => Some(Key::S),

117	            Eraser => Some(Key::E),

120	            Zoom => Some(Key::Z),
121	            // Animate leaves these two without a letter.
122	            GradientTransform | PolyStar => None,

126	    /// The button's glyph.
127	    ///
128	    /// # Why these are letters
129	    ///
130	    /// Adobe's tool icons cannot be used, and egui's bundled fonts cover only a
131	    /// scattering of the symbols a tool palette wants — measured on the real
132	    /// window, `▭ ◯ ╱ ➤ ⇱ ⤢ ✒ ▨` all rendered as empty boxes, which is worse
133	    /// than no icon at all.
134	    ///
135	    /// So each button shows its **shortcut letter**. It always renders, and it
136	    /// teaches the keyboard shortcut every time the user looks at the palette.
137	    /// The two tools Animate leaves without a letter use symbols confirmed to
138	    /// render. Full names are in the tooltip.
139	    pub fn glyph(self) -> &'static str {
140	        use ToolId::*;
141	        match self {
142	            Selection => "V",

146	            FreeTransform => "Q",

150	            Pen => "P",
151	            Text => "T",
152	            Line => "N",

170	    /// Is this tool implemented?
171	    pub fn status(self) -> ToolStatus {
172	        use ToolId::*;
173	        match self {
174	            Selection | Subselection | Lasso | MagicWand | FreeTransform | Line | Rectangle
175	            | Oval | PolyStar | Pencil | Brush | Eraser | PaintBucket | InkBottle | Eyedropper
176	            | Hand | Zoom | Pen | Camera | Bone | AssetWarp | MotionPath | GradientTransform
177	            | Text => ToolStatus::Ready,
178	        }
179	    }
180	
181	    pub fn is_ready(self) -> bool {
182	        matches!(self.status(), ToolStatus::Ready)
183	    }
184	
185	    /// Does this tool draw a new shape, rather than edit existing ones?
186	    pub fn is_drawing_tool(self) -> bool {
187	        use ToolId::*;
188	        matches!(
189	            self,
190	            Line | Rectangle | Oval | PolyStar | Pencil | Brush | Pen
191	        )
192	    }
193	
194	    /// Does the tool navigate rather than modify the document?
195	    ///
196	    /// Navigation tools stay usable on a locked layer.
197	    pub fn is_navigation(self) -> bool {
198	        matches!(self, ToolId::Hand | ToolId::Zoom)
199	    }
200	
201	    /// Does the tool edit the camera rather than artwork?
202	    ///
203	    /// The camera belongs to the document, so this is *not* navigation even
204	    /// though it looks like panning: moving it changes the exported result.
205	    pub fn is_camera(self) -> bool {
206	        matches!(self, ToolId::Camera)
207	    }
208	}

210	/// Toolbar groups, separated by a divider, following Animate's arrangement.
211	pub const TOOL_GROUPS: &[&[ToolId]] = &[
212	    &[

214	        ToolId::Subselection,

227	    &[ToolId::Bone, ToolId::AssetWarp, ToolId::MotionPath],
228	    &[
229	        ToolId::PaintBucket,
230	        ToolId::InkBottle,
231	        ToolId::Eyedropper,
232	        ToolId::Eraser,
233	    ],
234	    &[ToolId::Camera, ToolId::Hand, ToolId::Zoom],
235	];

237	/// Every tool, in toolbar order.
238	pub fn all_tools() -> Vec<ToolId> {
239	    TOOL_GROUPS.iter().flat_map(|g| g.iter().copied()).collect()
240	}

242	/// Which tool a bare letter selects.
243	pub fn tool_for_key(key: Key) -> Option<ToolId> {
244	    all_tools().into_iter().find(|t| t.shortcut() == Some(key))
245	}
